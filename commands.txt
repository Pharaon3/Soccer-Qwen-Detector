# Soccer Qwen Detector — commands
# Run from project root: Soccer-Qwen-Detector

# -----------------------------------------------------------------------------
# Setup
# -----------------------------------------------------------------------------

python -m venv .venv

# Windows PowerShell
.\.venv\Scripts\Activate.ps1

# Install (GPU PyTorch: see https://pytorch.org/get-started/locally/)
pip install -r requirements.txt

# Optional: 4-bit QLoRA (train_config.yaml → qwen.load_in_4bit: true)
# pip install bitsandbytes

# -----------------------------------------------------------------------------
# Help
# -----------------------------------------------------------------------------

python run_inference.py --help
python run_train.py --help
python convert_qwen_to_annotations.py --help
python run_eval.py --help

# -----------------------------------------------------------------------------
# A. Fine-tune Qwen2.5-VL (main training)
# -----------------------------------------------------------------------------
# Requires: dataset/<clip_id>/<clip_id>.json + video in each folder
# Edit train_config.yaml: video.sample_fps, video.max_frames (keep in sync with config.yaml)

python run_train.py --config train_config.yaml

# Quick test on a few clips
python run_train.py --max_clips 4 --epochs 1

# Override epochs
python run_train.py --epochs 5

# Checkpoints:
#   checkpoints/qwen_lora/          — final adapter (use for inference)
#   checkpoints/qwen_lora/epoch_* — last 2 kept (keep_last_n_checkpoints)

# -----------------------------------------------------------------------------
# B. Inference — fine-tuned Qwen (local Hugging Face, default)
# -----------------------------------------------------------------------------

python run_inference.py --backend hf --adapter checkpoints\qwen_lora --video data\videos\clip.mp4 --output outputs\clip.json

python run_inference.py --backend hf --video_dir data\videos --output_dir outputs\

# Uses config.yaml qwen.adapter_path if --adapter omitted
python run_inference.py --backend hf --video data\videos\clip.mp4 --output outputs\clip.json

# -----------------------------------------------------------------------------
# C. Inference — remote API (vLLM / OpenAI-compatible, no local adapter)
# -----------------------------------------------------------------------------

python run_inference.py --backend api --video data\videos\clip.mp4 --config config.yaml --output outputs_qwen\clip.json

python run_inference.py --backend api --video_dir data\videos --output_dir outputs_qwen\

# -----------------------------------------------------------------------------
# D. Convert API pseudo-labels → dataset annotations (optional)
# -----------------------------------------------------------------------------

python convert_qwen_to_annotations.py --input outputs_qwen\clip.json --output dataset\clip_id\clip_id.json --min_confidence 0.5

python convert_qwen_to_annotations.py --input_dir outputs_qwen\ --dataset_dir dataset\ --min_confidence 0.5

# -----------------------------------------------------------------------------
# E. Evaluate predictions vs ground truth
# -----------------------------------------------------------------------------

python run_eval.py --pred_dir outputs --gt_dir dataset --tolerance_sec 1.0

# Writes: outputs_eval\metrics.json  outputs_eval\errors.json

# -----------------------------------------------------------------------------
# Full pipeline (typical)
# -----------------------------------------------------------------------------
# 1. Annotate in dataset/annotator/  (or use API labels via D)
# 2. Place videos next to JSON under dataset/<clip_id>/
# 3. python run_train.py --config train_config.yaml
# 4. python run_inference.py --backend hf --adapter checkpoints\qwen_lora --video ... --output ...

# -----------------------------------------------------------------------------
# Video sampling (train_config.yaml + config.yaml — keep both aligned)
# -----------------------------------------------------------------------------
#   video.fps: 25              — timeline for timestamps (match source video)
#   video.sample_fps: 8        — frames sampled per second of clip (NOT native fps alone)
#   video.max_frames: 240       — hard cap (30s * 8fps = 240 = full clip at 8fps)
#   video.max_image_side: 1280  — downscale 1920x1080 before model
#
# sample_fps: 25 + max_frames: 750 is NOT practical (one prompt); use 2–8 fps + 60–240 frames.
# Two GPUs: device_map auto may help; still keep max_frames reasonable.

# -----------------------------------------------------------------------------
# Legacy — ResNet/CNN (optional, not Qwen)
# -----------------------------------------------------------------------------

python run_train_cnn.py --config train_config_cnn.yaml

python run_model_inference.py --video data\videos\clip.mp4 --checkpoint checkpoints\cnn\last.pt --config train_config_cnn.yaml --output outputs_model\clip.json

python tune_thresholds.py --checkpoint checkpoints\cnn\last.pt --config train_config_cnn.yaml --output thresholds.yaml

# -----------------------------------------------------------------------------
# Outputs (inference)
# -----------------------------------------------------------------------------
#   outputs\clip.json       — postprocessed events (source: qwen_finetuned or qwen_api)
#   outputs\clip_raw.txt    — raw model text
#   outputs\clip_error.txt  — on JSON parse failure

# -----------------------------------------------------------------------------
# Sanity check
# -----------------------------------------------------------------------------

python -m py_compile run_train.py run_inference.py
