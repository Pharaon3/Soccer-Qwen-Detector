# -----------------------------------------------------------------------------
# CLI arguments reference (run_inference.py)
# -----------------------------------------------------------------------------
#
# Source (mutually exclusive; exactly one required):
#   --video PATH          Single video file (.mp4, .avi, .mkv, .mov, .webm)
#   --video_dir PATH      Directory of video files to process
#
# Output (mutually exclusive; exactly one required):
#   --output PATH         Output JSON for single-video mode
#   --output_dir PATH     Output directory for batch mode
#
# Optional:
#   --config PATH         YAML config (default: config.yaml in project root)
#
# -----------------------------------------------------------------------------
# Outputs
# -----------------------------------------------------------------------------
#
# Single-video with --output outputs\clip.json:
#   outputs\clip.json       — postprocessed predictions
#   outputs\clip_raw.txt    — raw model response
#
# Batch with --output_dir outputs\:
#   outputs\{video_stem}.json
#   outputs\{video_stem}_raw.txt
#
# On JSON parse failure:
#   outputs\{stem}_error.txt — error + raw output (and process exits with error)
#
# -----------------------------------------------------------------------------
# Optional checks
# -----------------------------------------------------------------------------

python -m py_compile run_inference.py
